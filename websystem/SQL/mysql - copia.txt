CREATE TABLE menu_sinselect (
	id INT AUTO_INCREMENT PRIMARY KEY,
	nombre VARCHAR(255) NOT NULL,
	modulo VARCHAR(255) NOT NULL,   
	link VARCHAR(255) NOT NULL,
	Num_nivel VARCHAR(3),
	secciones VARCHAR(255),
	estilos VARCHAR(255),
	cod VARCHAR(10) UNIQUE,
	codtab VARCHAR(10)
);

-- Eliminar el trigger previo si existe
DROP TRIGGER IF EXISTS before_insert_cod_menu_sinselect;

-- Crear el trigger
DELIMITER $$

CREATE TRIGGER before_insert_cod_menu_sinselect
BEFORE INSERT ON menu_sinselect
FOR EACH ROW
BEGIN
    DECLARE max_cod INT;

    -- Obtener el valor máximo actual de la columna `cod` (numérico después del prefijo)
    SELECT IFNULL(MAX(CAST(SUBSTRING(cod, 4) AS UNSIGNED)), 0) + 1 INTO max_cod
    FROM menu_sinselect;

    -- Generar un nuevo valor para `cod` si está vacío o es NULL
    IF NEW.cod IS NULL OR NEW.cod = '' THEN
        SET NEW.cod = CONCAT('sin', LPAD(max_cod, 3, '0')); -- Prefijo 'sin' seguido de un número con 3 dígitos
    END IF;
END$$

DELIMITER ;
